JavaScript 程序结构

王秀宁 · 条件 & 循环

条件 1. if / else if / else

let wxn_score = 85;
// 根据分数判断等级
if (wxn_score >= 90) {
    wxn_grade = "优秀";
} else if (wxn_score >= 80) {
    wxn_grade = "良好";
} else if (wxn_score >= 60) {
    wxn_grade = "及格";
} else {
    wxn_grade = "不及格";
}

条件 2. switch（多分支）

let wxn_day = 3;  // 1=周一, 7=周日
switch (wxn_day) {
    case 1: wxn_week = "星期一"; break;
    case 2: wxn_week = "星期二"; break;
    case 3: wxn_week = "星期三"; break;
    case 4: wxn_week = "星期四"; break;
    default: wxn_week = "其它";
}

条件 3. 三元运算符 ? :

let wxn_age = 20;
let wxn_isAdult = wxn_age >= 18 ? "成年" : "未成年";

循环 1. for（次数循环）

// 计算 1+2+3+...+100
let wxn_sum = 0;
for (let wxn_i = 1; wxn_i <= 100; wxn_i++) {
    wxn_sum += wxn_i;
}

循环 2. for...of（遍历数组）

const wxn_fruits = ["🍎苹果", "🍌香蕉", "🍊橘子", "🍇葡萄"];
for (let wxn_fruit of wxn_fruits) {
    wxn_list.push(wxn_fruit);
}

循环 3. while（条件循环）

let wxn_count = 0;
let wxn_n = 1;
while (wxn_n < 100) {
    wxn_count++;
    wxn_n *= 2;  // 2, 4, 8, 16, 32, 64, 128...
}

循环 4. do...while（至少执行一次）

let wxn_doCount = 0;
do {
    wxn_doCount++;
} while (wxn_doCount < 3);

循环 5. for...in（遍历对象键）

let wxn_user = { name: "王秀宁", age: 20, city: "大连" };
let wxn_keys = [];
for (let wxn_k in wxn_user) {
    wxn_keys.push(wxn_k + "=" + wxn_user[wxn_k]);
}

循环 break 与 continue

// break: 跳出整个循环；continue: 跳过当前一轮
let wxn_evens = [];
for (let wxn_i = 1; wxn_i <= 10; wxn_i++) {
    if (wxn_i === 8) break;       // 到 8 就停
    if (wxn_i % 2 !== 0) continue;  // 奇数跳过
    wxn_evens.push(wxn_i);
}
